cmake_minimum_required(VERSION 3.16)
project(ArenaGlassSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.10 REQUIRED COMPONENTS Widgets Network)
find_package(ZLIB REQUIRED)
qt5_add_resources(ICONS ../arenaicons.qrc)
add_executable(arena-glass-smoke test_glass.cpp ${ICONS})
target_link_libraries(arena-glass-smoke PRIVATE Qt5::Widgets Qt5::Network ZLIB::ZLIB)
if(WIN32)
  target_link_libraries(arena-glass-smoke PRIVATE winhttp)
endif()
if(MSVC)
  target_compile_options(arena-glass-smoke PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME arena-glass-smoke COMMAND arena-glass-smoke)
set_tests_properties(arena-glass-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
